5. Replace handleAddToCart

Your current version simply calls addToFashionCart().

Replace the entire function with:

function handleAddToCart(
  design: FashionDesign,
  size: FashionSize,
) {
  const stock = Number(
    size.stock_quantity,
  )

  const currentQuantity =
    getCartQuantityForSize(
      design.id,
      size.id,
    )

  if (
    !size.is_active ||
    stock <= 0
  ) {
    return
  }

  if (
    currentQuantity >= stock
  ) {
    return
  }

  const cart =
    addToFashionCart({
      designId: design.id,
      designSizeId: size.id,
      size: size.size,
      quantity: 1,
    })

  syncCart(cart)
}

function handleIncreaseCartItem(
  item: FashionCartItem,
) {
  const design =
    designs.find(
      (entry) =>
        entry.id ===
        item.designId,
    )

  if (!design) {
    return
  }

  const size =
    design.sizes.find(
      (entry) =>
        entry.id ===
        item.designSizeId,
    )

  if (!size) {
    return
  }

  const stock = Number(
    size.stock_quantity,
  )

  const currentQuantity =
    Number(item.quantity)

  if (
    !size.is_active ||
    stock <= 0
  ) {
    return
  }

  if (
    currentQuantity >= stock
  ) {
    return
  }

  const cart =
    updateFashionCartQuantity(
      item.designId,
      item.designSizeId,
      currentQuantity + 1,
    )

  syncCart(cart)
}

function FashionProductDetailsModal({
  design,
  offer,
  onClose,
  onAddToCart,
  onBookNow,
  isInCart,
}: {
  design: FashionDesign
  offer: FashionOffer | null
  onClose: () => void
  onAddToCart: (
    design: FashionDesign,
    size: FashionSize,
  ) => void
  onBookNow: (
    design: FashionDesign,
    size: FashionSize,
  ) => void
  isInCart: (
    designId: string,
    sizeId: string,

}) {
  const activeSizes =
    getActiveSizes(
      design.sizes,
    )

  const [
    selectedSizeId,
    setSelectedSizeId,
  ] = useState(
    activeSizes[0]?.id ?? '',
  )

  const [
    currentImageIndex,
    setCurrentImageIndex,
  ] = useState(0)

  const images =
    getSortedImages(
      design.images,
    )

  const selectedSize =
    activeSizes.find(
      (size) =>
        size.id ===
        selectedSizeId,
    ) ??
    activeSizes[0] ??
    null

  useEffect(() => {
    setSelectedSizeId(
      activeSizes[0]?.id ?? '',
    )

    setCurrentImageIndex(0)
  }, [design.id])

  useEffect(() => {
    document.body.style.overflow =
      'hidden'

    return () => {
      document.body.style.overflow =
        ''
    }
  }, [])

  useEffect(() => {
    function handleEscape(
      event: KeyboardEvent,
    ) {
      if (
        event.key === 'Escape'
      ) {
        onClose()
      }
    }

    document.addEventListener(
      'keydown',
      handleEscape,
    )

    return () => {
      document.removeEventListener(
        'keydown',
        handleEscape,
      )
    }
  }, [onClose])

  const basePrice =
    selectedSize
      ? Number(
          selectedSize.price,
        )
      : null

  const discountedPrice =
    basePrice !== null
      ? calculateDiscountedPrice(
          basePrice,
          offer,
        )
      : null

  const hasDiscount =
    basePrice !== null &&
    discountedPrice !== null &&
    discountedPrice <
      basePrice

  const stock =
    selectedSize
      ? Number(
          selectedSize.stock_quantity,
        )
      : 0

  const isAvailable =
    Boolean(
      selectedSize &&
        selectedSize.is_active &&
        stock > 0,
    )

  const selectedSizeInCart =
    selectedSize
      ? isInCart(
          design.id,
          selectedSize.id,
        )
      : false

  function nextImage() {
    if (!images.length) {
      return
    }

    setCurrentImageIndex(
      (current) =>
        (current + 1) %
        images.length,

  function previousImage() {
    if (!images.length) {
      return
    }

          : current - 1,

  }

  return (

      className="fashion-product-modal-backdrop"
      onMouseDown={(event) => {
        if (
          event.target ===
          event.currentTarget
        ) {
          onClose()
        }
      }}
    >
      <div
        className="fashion-product-modal"
        role="dialog"
        aria-modal="true"
        aria-label={`${design.name} details`}
      >
        <header className="fashion-product-modal-header">
          <div>
            <span>
              FASHION / PRODUCT
            </span>

            <h2>
              {design.name}
            </h2>
          </div>

          <button
            type="button"
            className="fashion-product-modal-close"
            onClick={onClose}
            aria-label="Close product details"
          >
            ×
          </button>
        </header>

        <div className="fashion-product-modal-body">

          {/* =================================================
              IMAGE GALLERY
          ================================================= */}

          <section className="fashion-product-gallery">

                  <button
                    key={image.id}
                    type="button"
                    className={
                      index ===
                      currentImageIndex
                        ? 'active'
                        : ''
                    }
                    onClick={() =>
                      setCurrentImageIndex(
                        index,
                      )
                    }

                    <img
                      src={
                        getImageUrl(
                          image.image_url,
                        ) ?? ''
                      }
                      alt={
                        image.alt_text ??
                        design.name
                      }
                    />
                  </button>

            </div>

            <div className="fashion-product-main-image">

                currentImageIndex
              ] ? (
                <img
                  src={
                    getImageUrl(
                      images[
                        currentImageIndex
                      ]
                        .image_url,

                  }

                      currentImageIndex
                    ].alt_text ??

                />
              ) : (

                  W

                  {hasDiscount &&
                    offer && (
                      <span>
                        {getDiscountText(
                          offer,
                        )}
                      </span>
                    )}
                </>
              ) : (
                <strong>
                  Price on request
                </strong>
              )}
            </div>

            {/* SIZE */}

            <div className="fashion-product-size-section">

              <div>
                <strong>
                  Select Size
                </strong>

                {selectedSize && (
                  <span>
                    {selectedSize.size}
                  </span>
                )}
              </div>

              <div className="fashion-product-size-list">

                {activeSizes.map(
                  (size) => {
                    const sizeStock =
                      Number(
                        size.stock_quantity,
                      )

                    return (
                      <button
                        key={size.id}
                        type="button"
                        disabled={
                          sizeStock <= 0
                        }
                        className={
                          selectedSizeId ===
                          size.id
                            ? 'active'
                            : ''
                        }
                        onClick={() =>
                          setSelectedSizeId(
                            size.id,
                          )
                        }
                      >
                        <span>
                          {size.size}
                        </span>

                        {sizeStock <=
                        0 && (
                          <small>
                            Out
                          </small>
                        )}
                      </button>
                    )
                  },
                )}

              </div>

            </div>

            {/* STOCK */}

            {selectedSize && (
              <div
                className={`fashion-product-stock ${
                  stock <= 0
                    ? 'out'
                    : stock <= 5
                      ? 'low'
                      : ''
                }`}
              >
                {stock <= 0
                  ? 'Out of stock'
                  : stock <= 5
                    ? `Only ${stock} left in stock`
                    : `${stock} available in stock`}
              </div>
            )}

            {/* DELIVERY */}

            <div className="fashion-product-delivery">
              <CheckIcon />

              <span>
                Delivery:{' '}
                <strong>
                  {
                    design.delivery_min_days
                  }
                  –
                  {
                    design.delivery_max_days
                  } days
                </strong>
              </span>
            </div>

            {/* ACTIONS */}

            <div className="fashion-product-actions">

              <button
                type="button"
                disabled={
                  !selectedSize ||
                  !isAvailable ||
                  selectedSizeInCart
                }
                onClick={() => {
                  if (
                    selectedSize &&
                    isAvailable &&
                    !selectedSizeInCart
                  ) {
                    onAddToCart(
                      design,
                      selectedSize,
                    )
                  }
                }}
              >
                {selectedSizeInCart
                  ? 'Already in Cart'
                  : 'Add to Cart'}
              </button>

              <button
                type="button"
                disabled={
                  !selectedSize ||
                  !isAvailable
                }
                onClick={() => {
                  if (
                    selectedSize &&
                    isAvailable
                  ) {
                    onBookNow(
                      design,
                      selectedSize,
                    )

                    onClose()
                  }
                }}
              >
                Buy Now
              </button>

            </div>

            {/* =================================================
                ABOUT THIS ITEM
            ================================================= */}

            {design.product_details
              ?.trim() && (
              <section className="fashion-product-info-section">

                <h2>
                  About this item

                <div className="fashion-product-text">

One important stock point

This fixes the frontend cart quantity limit, so if stock is 5, the customer cannot use the cart + button to reach 6 or 7.

However, this is not yet the final inventory guarantee. A production ecommerce system must also validate/decrement stock server-side when the order is created, because two customers could both see 5 available at the same time.

We should do that in the Fashion Booking / order creation file next. Your current booking page also has quantity controls, so that is the next place we should fix after this page.
